/**
 * Cihazın baktığı yönü (heading) ölçer.
 * - iOS: DeviceOrientationEvent.requestPermission() gerekir (kullanıcı jesti ile)
 * - Android: doğrudan deviceorientationabsolute çalışır
 * - Masaüstü: desteklenmez, null döner
 *
 * Heading = 0 → kuzey, 90 → doğu, 180 → güney, 270 → batı
 */

export type CompassCallback = (heading: number) => void;

export interface CompassHandle {
  stop: () => void;
  isActive: () => boolean;
}

type IOSDeviceOrientationEvent = typeof DeviceOrientationEvent & {
  requestPermission?: () => Promise<'granted' | 'denied'>;
};

export function isCompassSupported(): boolean {
  if (typeof window === 'undefined') return false;
  return 'DeviceOrientationEvent' in window;
}

/**
 * iOS 13+ için izin ister. Android'de doğrudan 'granted' döner.
 * Kullanıcı jesti (buton tıklaması) içinde çağrılmalı.
 */
export async function requestCompassPermission(): Promise<boolean> {
  if (!isCompassSupported()) return false;

  const DOE = window.DeviceOrientationEvent as IOSDeviceOrientationEvent;
  if (typeof DOE.requestPermission === 'function') {
    try {
      const result = await DOE.requestPermission();
      return result === 'granted';
    } catch {
      return false;
    }
  }
  // Android ve masaüstü: izin istemez
  return true;
}

/**
 * Pusulayı başlatır. Her yön değişiminde callback çağrılır.
 * Not: callback sık çağrılır (saniyede ~10 kez) — debounce çağıranda.
 */
export function startCompass(
  callback: CompassCallback,
  onError?: (reason: string) => void
): CompassHandle {
  if (!isCompassSupported()) {
    onError?.('unsupported');
    return { stop: () => {}, isActive: () => false };
  }

  let active = true;

  const handler = (e: DeviceOrientationEvent) => {
    if (!active) return;

    // iOS: webkitCompassHeading (0-360, saat yönünde)
    const webkit = (e as DeviceOrientationEvent & { webkitCompassHeading?: number })
      .webkitCompassHeading;
    if (typeof webkit === 'number' && !Number.isNaN(webkit)) {
      callback(normalize(webkit));
      return;
    }

    // Android: yalnızca MUTLAK (kuzeye göre) yönelim güvenilir; göreli alpha yanlış yön verir
    const absolute =
      (e as DeviceOrientationEvent & { absolute?: boolean }).absolute === true;
    if (absolute && typeof e.alpha === 'number' && !Number.isNaN(e.alpha)) {
      callback(headingFromOrientation(e.alpha, e.beta ?? 0, e.gamma ?? 0));
      return;
    }

    onError?.('no-heading');
  };

  // Absolute varsa onu tercih et
  const useAbsolute = 'ondeviceorientationabsolute' in window;
  const eventName = useAbsolute ? 'deviceorientationabsolute' : 'deviceorientation';

  window.addEventListener(eventName, handler as EventListener, true);

  return {
    stop: () => {
      active = false;
      window.removeEventListener(eventName, handler as EventListener, true);
    },
    isActive: () => active,
  };
}

/**
 * alpha/beta/gamma (W3C, mutlak) → pusula yönü (0=kuzey, saat yönünde).
 * Telefon yataya yakınsa üst kenarın, dik tutuluyorsa arka yüzün (kamera) baktığı yönü verir.
 * Böylece yalnızca "360 - alpha" kullanıldığında dik tutulan telefonda oluşan sapma önlenir.
 */
export function headingFromOrientation(alpha: number, beta: number, gamma: number): number {
  const d = Math.PI / 180;
  const a = alpha * d;
  const b = beta * d;
  const g = gamma * d;
  const cA = Math.cos(a), sA = Math.sin(a);
  const cB = Math.cos(b), sB = Math.sin(b);
  const cG = Math.cos(g), sG = Math.sin(g);

  const facingUp = Math.abs(cB * cG); // 1 = tam yatay, 0 = dik
  let east: number;
  let north: number;
  if (facingUp > 0.7) {
    // yatay: cihazın üst kenarı
    east = -sA * cB;
    north = cA * cB;
  } else {
    // dik: arka yüz
    east = -cA * sG - sA * sB * cG;
    north = -sA * sG + cA * sB * cG;
  }
  return normalize((Math.atan2(east, north) * 180) / Math.PI);
}

function normalize(deg: number): number {
  return ((deg % 360) + 360) % 360;
}

import type { SavedGate, NavUpdate } from './types';
import { findDoor, doorLabel } from './doors';
import { distanceM, bearingDeg } from './trail';

const ARRIVE_RADIUS_M = 20;
const MAX_ALLOWED_DISTANCE_M = 2000;
const MAX_ARRIVE_RADIUS_M = 40;

export interface NavResult extends NavUpdate {
  isTooFar: boolean;
  /** Konum kullanıcı ölçümü değil, doğrulanmamış katalogdan geliyorsa true */
  approximate: boolean;
  /** Pusula varsa: hedef yönü ile cihaz yönü arasındaki fark (-180..180) */
  relativeBearing?: number;
}

/**
 * Hedef HER ZAMAN kullanıcının kapının önündeyken kaydettiği konumdur.
 * (Katalog koordinatları doğrulanmadığı için hedef olarak kullanılmaz.)
 */
export function resolveTarget(saved: SavedGate): { lat: number; lng: number; name: string } {
  const door = findDoor(saved.doorId);
  return {
    lat: saved.outsideLat,
    lng: saved.outsideLng,
    name: door?.name ?? saved.doorName,
  };
}

function arriveRadius(accuracyM?: number | null): number {
  if (typeof accuracyM !== 'number' || !Number.isFinite(accuracyM)) return ARRIVE_RADIUS_M;
  return Math.min(Math.max(accuracyM, ARRIVE_RADIUS_M), MAX_ARRIVE_RADIUS_M);
}

/** 2,5 km / 250 km biçiminde (Türkçe ondalık virgül) */
export function formatKm(meters: number): string {
  const km = meters / 1000;
  if (km >= 10) return `${Math.round(km)} km`;
  return `${km.toFixed(1).replace('.', ',')} km`;
}

function formatDistance(d: number): string {
  if (d < 50) return `${Math.round(d)} metre`;
  const r = Math.round(d / 10) * 10;
  if (r < 1000) return `${r} metre`;
  return `${(d / 1000).toFixed(1)} kilometre`;
}

export function computeNavigation(
  saved: SavedGate,
  current: { lat: number; lng: number },
  deviceHeading?: number | null,
  accuracyM?: number | null
): NavResult {
  const { lat: targetLat, lng: targetLng, name: doorName } = resolveTarget(saved);
  const approximate = saved.locationSource !== 'gps';

  const distance = distanceM(current, { lat: targetLat, lng: targetLng });
  const bearingDegVal = bearingDeg(current, { lat: targetLat, lng: targetLng });

  // relative bearing: hedef yönü - cihaz yönü
  const relativeBearing =
    typeof deviceHeading === 'number' && !Number.isNaN(deviceHeading)
      ? normalizeSigned(bearingDegVal - deviceHeading)
      : undefined;

  // Kayıtlı (katalogdaki) kapı: Mescid-i Haram'a uzaklık, kapının gerçek konumundan ölçülür.
  // Kullanıcının kendi yazdığı kapı: önceki davranış (kayıtlı noktaya uzaklık).
  const catalogDoor = findDoor(saved.doorId);
  if (catalogDoor) {
    const farDistance = distanceM(current, {
      lat: catalogDoor.outsideLat,
      lng: catalogDoor.outsideLng,
    });
    if (farDistance > MAX_ALLOWED_DISTANCE_M) {
      return {
        distance: farDistance,
        bearingDeg: bearingDeg(current, { lat: catalogDoor.outsideLat, lng: catalogDoor.outsideLng }),
        arrived: false,
        isTooFar: true,
        approximate,
        relativeBearing,
        instruction:
          `${catalogDoor.name} Kapısı'ndan kuş uçuşu yaklaşık ${formatKm(farDistance)} uzaktasınız. ` +
          "Bu mesafe yürümek için uygun değil. Mescid-i Haram'a yaklaşınca tekrar deneyin.",
      };
    }
  } else if (distance > MAX_ALLOWED_DISTANCE_M) {
    return {
      distance,
      bearingDeg: bearingDegVal,
      arrived: false,
      isTooFar: true,
      approximate,
      relativeBearing,
      instruction:
        "Mescid-i Haram'ın dışındasınız. Lütfen Harem bölgesine yaklaşıp tekrar deneyin.",
    };
  }

  if (distance < arriveRadius(accuracyM)) {
    return {
      distance,
      bearingDeg: bearingDegVal,
      arrived: true,
      isTooFar: false,
      approximate,
      relativeBearing,
      instruction:
        `Giriş kapınızın önündesiniz (${doorLabel(saved.doorId, doorName)}). Hedefe ulaştınız.` +
        (approximate ? ' Kapı numarasını tabeladan kontrol edin.' : ''),
    };
  }

  const distStr = formatDistance(distance);

  // Pusula varsa göreli yön tarifi, yoksa kuzey referanslı
  const dirInstruction =
    relativeBearing !== undefined
      ? relativeTurnInstruction(relativeBearing)
      : `${bearingToTurkish(bearingDegVal)} yönünde yürüyün`;

  return {
    distance,
    bearingDeg: bearingDegVal,
    arrived: false,
    isTooFar: false,
    approximate,
    relativeBearing,
    instruction: `${distStr} ${dirInstruction}.`,
  };
}

/**
 * Göreli yöne göre "düz yürü / sağa dön / arkanı dön" tarifi üretir.
 * -45..45 → düz yürü
 * 45..135 → sağa dön
 * 135..180 veya -180..-135 → arkanı dön
 * -135..-45 → sola dön
 */
function relativeTurnInstruction(relative: number): string {
  const abs = Math.abs(relative);

  if (abs <= 20) return 'düz yürüyün';
  if (abs <= 45) {
    return relative > 0 ? 'hafifçe sağa dönüp yürüyün' : 'hafifçe sola dönüp yürüyün';
  }
  if (abs <= 135) {
    return relative > 0 ? 'sağa dönüp yürüyün' : 'sola dönüp yürüyün';
  }
  return 'arkanıza dönüp yürüyün';
}

function normalizeSigned(deg: number): number {
  // -180..180 aralığına indir
  let d = ((deg + 180) % 360 + 360) % 360 - 180;
  if (d === -180) d = 180;
  return d;
}

function bearingToTurkish(b: number): string {
  const sectors = [
    'kuzey', 'kuzeydoğu', 'doğu', 'güneydoğu',
    'güney', 'güneybatı', 'batı', 'kuzeybatı',
  ] as const;
  return sectors[Math.round(b / 45) % 8]!;
}

import { findDoor } from './doors';

export interface HelpMessages {
  arabicTitle: string;
  arabicBody: string;
  turkishBody: string;
  /** Hoparlörden okunacak Arapça metin (başlık + gövde) */
  speakArabic: string;
  /** Kullanıcıya Türkçe söylenecek yönerge */
  speakTurkish: string;
}

const DOOR_AR: Record<number, string> = {
  1: 'باب الملك عبد العزيز',
  79: 'باب الملك فهد',
  100: 'باب الملك عبد الله',
  62: 'باب العمرة',
  45: 'باب الفتح',
  19: 'باب السلام',
  13: 'باب الصفا',
  64: 'جسر الشبيكة',
};

const TITLE_AR = 'أحتاج مساعدة';
const SPEAK_TR = 'Bu ekranı görevliye gösterin.';

export function getHelpMessages(saved: {
  doorId: number;
  doorName: string;
} | null): HelpMessages {
  // Kayıt yoksa genel yardım mesajı
  if (!saved) {
    const body = 'أنا تائه. هل يمكنك أن تساعدني؟';
    return {
      arabicTitle: TITLE_AR,
      arabicBody: body,
      turkishBody: 'Yolumu kaybettim. Bana yardım edebilir misiniz?',
      speakArabic: `${TITLE_AR}. ${body}`,
      speakTurkish: SPEAK_TR,
    };
  }

  // Kapı numarası evrenseldir: ad çevirisi yanlış/eksik olsa bile görevli numarayı anlar.
  const arName = DOOR_AR[saved.doorId];
  const target = arName
    ? `${arName}، الباب رقم ${saved.doorId}`
    : `الباب رقم ${saved.doorId}`;
  const arabicBody = `أريد الذهاب إلى ${target}. هل يمكنك أن تخبرني في أي جهة يقع؟`;
  const turkishTarget = saved.doorName
    ? `${saved.doorId} numaralı ${saved.doorName} Kapısı'na`
    : `${saved.doorId} numaralı kapıya`;
  const turkishBody =
    `${turkishTarget} gitmek istiyorum. Ne tarafta olduğunu söyler misiniz?`;

  return {
    arabicTitle: TITLE_AR,
    arabicBody,
    turkishBody,
    speakArabic: `${TITLE_AR}. ${arabicBody}`,
    speakTurkish: SPEAK_TR,
  };
}

export function getArabicDoorName(doorId: number): string | null {
  return DOOR_AR[doorId] ?? null;
}

export function hasArabicTranslation(doorId: number): boolean {
  return doorId in DOOR_AR;
}

function escapeHtml(s: string): string {
  return s.replace(/[&<>"']/g, c =>
    ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c] as string)
  );
}

async function onSaveConfirm(): Promise<void> {
  if (pendingDoor === null) return;
  const door = pendingDoor;
  const catalog = findDoor(door.id);

  const token = ++saveToken;
  const retry = document.getElementById('btnSaveRetry');
  retry?.classList.add('hide');
  show('saving');
  const fix = document.getElementById('fix');
  if (fix) fix.textContent = 'Konum aranıyor…';
  speak('Kapı kaydediliyor. Lütfen birkaç saniye bekleyin.', true);

  // Doğrulanmamış katalog koordinatı ASLA kaydedilmez: yalnızca kullanıcının
  // kapının önündeyken ölçülen gerçek GPS konumu. Yoksa kayıt yapılmaz.
  let pos: GeolocationPosition | null = null;
  let errCode = 0;
  for (let attempt = 0; attempt < 2; attempt++) {
    try {
      const p = await getCurrentPosition(10000);
      if (token !== saveToken) return;
      if (!pos || p.coords.accuracy < pos.coords.accuracy) pos = p;
      if (p.coords.accuracy <= 30) break;
    } catch (e) {
      if (token !== saveToken) return;
      errCode = (e as { code?: number })?.code ?? 0;
      if (errCode === 1) break; // izin reddedildi, tekrar denemenin anlamı yok
    }
  }
  if (token !== saveToken) return;

  if (!pos || pos.coords.accuracy > MAX_SAVE_ACCURACY_M) {
    const msg =
      errCode === 1
        ? 'Konum izni verilmedi. Tarayıcı ayarlarından konum iznini açın.'
        : 'Konum hassas alınamadı. Açık alana çıkıp tekrar deneyin.';
    if (fix) fix.textContent = msg;
    retry?.classList.remove('hide');
    speak(msg, true);
    return;
  }

  // Kayıtlı (katalog) kapı: Mescid-i Haram'dan 2 km'den uzakta kayıt yapılmaz.
  // Kullanıcının kendi yazdığı kapıda yalnızca ölçülen konum esastır (eski davranış).
  const gateDistance = catalog
    ? distanceM(
        { lat: pos.coords.latitude, lng: pos.coords.longitude },
        { lat: catalog.outsideLat, lng: catalog.outsideLng }
      )
    : null;
  if (catalog && gateDistance !== null && gateDistance > MAX_SAVE_DISTANCE_M) {
    const msg =
      `Bulunduğunuz konum ${catalog.name} Kapısı'ndan ${formatKm(gateDistance)} uzakta. ` +
      "Lütfen Mescid-i Haram'a yaklaşınca kaydedin.";
    if (fix) fix.textContent = msg;
    retry?.classList.remove('hide');
    speak(msg, true);
    return;
  }

  // Seçilen numara, ölçülen konumla uyuşuyor mu? (katalog yalnızca bu denetim için)
  const mismatch = gateDistance !== null && gateDistance > MISMATCH_DISTANCE_M;

  recorder.stop();
  recorder.start();

  savedGate = {
    doorId: door.id,
    doorName: door.name,
    outsideLat: pos.coords.latitude,
    outsideLng: pos.coords.longitude,
    savedAt: Date.now(),
    trail: [],
    locationSource: 'gps',
    accuracy: pos.coords.accuracy,
    numberMismatch: mismatch,
  };
  saveGate(savedGate);
  renderSavedCard();
  if (mismatch) {
    speak(
      'Konumunuz kaydedildi, ancak seçtiğiniz kapı numarası bulunduğunuz yerle uyuşmuyor olabilir. Kapının üzerindeki numarayı kontrol edin.',
      true
    );
  } else {
    speak(`${doorLabel(door.id, door.name)} kaydedildi. İyi ziyaretler.`, true);
  }
  show('home');
}

function getCurrentPosition(timeoutMs: number): Promise<GeolocationPosition> {
  return new Promise((resolve, reject) => {
    if (!navigator.geolocation) return reject(new Error('no-geo'));
    navigator.geolocation.getCurrentPosition(resolve, reject, {
      enableHighAccuracy: true,
      maximumAge: 0,
      timeout: timeoutMs,
    });
  });
}

// ---------- Guide flow ----------
async function onGuide(): Promise<void> {
  if (guideStarting) return;
  // iOS: pusula izni kullanıcı jesti içinde, İLK await'ten önce istenmeli
  const compassPermission = requestCompassPermission().catch(() => false);

  savedGate = loadGate();
  if (!savedGate) {
    speak('Önce giriş kapınızı kaydedin.', true);
    show('home');
    return;
  }
  guideStarting = true;

  try {
    const trail = recorder.stop();
    if (trail.length > 0) {
      savedGate.trail = trail;
      saveGate(savedGate);
    }

    hasWarnedTooFar = false;
    hasFittedMap = false;
    lastAccuracy = null;
    lastBearing = null;
    gpsHeading = null;
    arrowState = 'idle';
    updateArrow();

    show('guide');
    initMap('map');
    setTimeout(() => invalidateSize(), 50);

    // Konum doğrulanmamışsa (eski kayıt) açıkça uyar
    const warn = document.getElementById('warn');
    if (warn) {
      const approx = savedGate.locationSource !== 'gps';
      warn.classList.toggle('hide', !approx);
      warn.textContent = approx
        ? '⚠️ Kayıtlı konum yaklaşık. Kapı numarasını tabeladan kontrol edin.'
        : '';
    }

    await keepAwake();

    // Pusula
    compass?.stop();
    compass = null;
    deviceHeading = null;
    if (await compassPermission) {
      compass = startCompass(
        onHeading,
        () => {
          // Pusula olmadan da çalışır
        }
      );
    }

    speak('Konumunuz alınıyor. Lütfen bekleyin.', true);

    try {
      await getCurrentPosition(8000);
    } catch {}

    const target = resolveTarget(savedGate);
    setTarget(target.lat, target.lng, doorLabel(savedGate.doorId, target.name));
    drawTrail(savedGate.trail);

    if (guideWatchId !== null) {
      navigator.geolocation.clearWatch(guideWatchId);
    }
    guideWatchId = navigator.geolocation.watchPosition(
      pos => onGuidePos(pos, target.lat, target.lng),
      err => onGuideErr(err),
      { enableHighAccuracy: true, maximumAge: 0, timeout: 20000 }
    );

    const gm = document.getElementById('gm') as HTMLAnchorElement | null;
    if (gm) {
      gm.href = `https://www.google.com/maps/dir/?api=1&destination=${target.lat},${target.lng}&travelmode=walking`;
    }
  } finally {
    guideStarting = false;
  }
}

function onGuidePos(pos: GeolocationPosition, tLat: number, tLng: number): void {
  if (!savedGate) return;
  const cur = { lat: pos.coords.latitude, lng: pos.coords.longitude };
  lastAccuracy = pos.coords.accuracy;

  // Yürürken GPS yönü (pusula yoksa veya şaşarsa yedek)
  const hd = pos.coords.heading;
  const spd = pos.coords.speed;
  if (typeof hd === 'number' && Number.isFinite(hd) && typeof spd === 'number' && spd >= 0.8) {
    gpsHeading = hd;
    gpsHeadingAt = Date.now();
  }

  setCurrent(cur.lat, cur.lng);

  const nav: NavResult = computeNavigation(savedGate, cur, effectiveHeading()?.value ?? null, lastAccuracy);
  drawLineToTarget(cur, { lat: tLat, lng: tLng }, nav.isTooFar ? formatKm(nav.distance) : undefined);

  if (!hasFittedMap) {
    fitToAll();
    hasFittedMap = true;
  }

  lastBearing = nav.bearingDeg;
  arrowState = nav.arrived ? 'arrived' : nav.isTooFar ? 'far' : 'go';
  updateArrow();
  const say = document.getElementById('say');
  const dist = document.getElementById('dist');

  if (say) say.textContent = nav.instruction;

  if (dist) {
    const weak = lastAccuracy > 100 ? ' (GPS zayıf)' : '';
    if (nav.isTooFar) {
      dist.textContent = `Uzaklık: ${formatKm(nav.distance)}`;
      dist.style.color = 'var(--r)';
    } else {
      dist.textContent =
        (nav.distance < 1000
          ? `${Math.round(nav.distance)} metre`
          : `${(nav.distance / 1000).toFixed(1)} km`) + weak;
      dist.style.color = '';
    }
  }

  if (nav.isTooFar) {
    if (!hasWarnedTooFar) {
      speak(nav.instruction, true);
      hasWarnedTooFar = true;
    }
    return;
  }
  hasWarnedTooFar = false;

  speak(nav.instruction, nav.arrived);

  if (nav.arrived && guideWatchId !== null) {
    navigator.geolocation.clearWatch(guideWatchId);
    guideWatchId = null;
    fitToAll();
    try {
      navigator.vibrate?.([200, 100, 200]);
    } catch {}
    if (say) say.style.color = 'var(--g)';
    if (dist) {
      dist.textContent = '✅ VARDINIZ';
      dist.style.color = 'var(--g)';
    }
    stopCompass();
    releaseAwake();
  }
}

/** Pusula yönünü yumuşat (titremeyi azaltır; 359°→1° geçişinde doğru çalışır) */
function onHeading(h: number): void {
  if (deviceHeading === null) {
    deviceHeading = h;
  } else {
    const d = ((h - deviceHeading + 540) % 360) - 180;
    deviceHeading = (deviceHeading + 0.3 * d + 360) % 360;
  }

}

function effectiveHeading(): { value: number; source: 'compass' | 'gps' } | null {
  if (deviceHeading !== null) return { value: deviceHeading, source: 'compass' };
  if (gpsHeading !== null && Date.now() - gpsHeadingAt < 8000) {
    return { value: gpsHeading, source: 'gps' };
  }
  return null;
}

function turnWord(rel: number): { text: string; tone: 'ok' | 'turn' | 'back' } {
  const abs = Math.abs(rel);
  if (abs <= 20) return { text: 'DÜZ İLERLEYİN', tone: 'ok' };
  if (abs <= 45) return { text: rel > 0 ? 'HAFİF SAĞA' : 'HAFİF SOLA', tone: 'turn' };
  if (abs <= 135) return { text: rel > 0 ? 'SAĞA DÖNÜN' : 'SOLA DÖNÜN', tone: 'turn' };
  return { text: 'ARKANIZA DÖNÜN', tone: 'back' };
}

function updateArrow(): void {
  const wrap = document.getElementById('arrowWrap');
  const svg = document.getElementById('arrow');
  const word = document.getElementById('arrowWord');
  const note = document.getElementById('arrowNote');
  if (!wrap || !svg) return;

  const heading = effectiveHeading();
  let state: string = arrowState;
  if (arrowState === 'go' && (heading === null || lastBearing === null)) state = 'nohead';
  wrap.setAttribute('data-state', state);

  if (state === 'go' && heading !== null && lastBearing !== null) {
    const rel = ((lastBearing - heading.value + 540) % 360) - 180; // -180..180
    const delta = ((((rel - arrowRot) % 360) + 540) % 360) - 180; // en kısa dönüş
    arrowRot += delta;
    svg.style.transform = `rotate(${arrowRot.toFixed(0)}deg)`;
    const w = turnWord(rel);
    wrap.setAttribute('data-tone', w.tone);
    if (word) word.textContent = w.text;
    if (note) {
      note.textContent =
        heading.source === 'compass'
          ? '🧭 Ok, telefonun baktığı yöne göre gösterir'
          : '📍 Ok, yürüdüğünüz yöne göre gösterir';
    }
    return;
  }

  svg.style.transform = '';
  if (state === 'nohead') {
    if (word) word.textContent = 'YÖN BULUNUYOR';
    if (note) note.textContent = 'Telefonu yatay tutun veya birkaç adım yürüyün';
  } else if (state === 'arrived') {
    wrap.setAttribute('data-tone', 'ok');
    if (word) word.textContent = 'VARDINIZ';
    if (note) note.textContent = '';
  } else {
    if (word) word.textContent = '';
    if (note) note.textContent = '';
  }
}

// ---------- Wake Lock ----------
async function keepAwake(): Promise<void> {
  try {
    if ('wakeLock' in navigator && !wakeLock) {
      wakeLock = await (navigator as Navigator & { wakeLock: WakeLock }).wakeLock.request('screen');
      wakeLock.addEventListener('release', () => {
        wakeLock = null;
      });
    }
  } catch {}
}

function releaseAwake(): void {
  if (wakeLock) {
    try {
      wakeLock.release();
    } catch {}
    wakeLock = null;
  }
}

let lastSpeak = 0;
let lastFull = '';
let lastSpokenText = '';
let voicesReady = false;

// Chrome'da utterance'lar çöp toplayıcıya kaybolup konuşma yarıda kesilebiliyor
const alive = new Set<SpeechSynthesisUtterance>();

const MIN_GAP_MS = 6000;

function allVoices(): SpeechSynthesisVoice[] {
  try {
    return speechSynthesis.getVoices() ?? [];
  } catch {
    return [];
  }
}

function pickVoice(lang: string): SpeechSynthesisVoice | undefined {
  const vs = allVoices();
  const lower = lang.toLowerCase();
  const base = lower.split('-')[0]!;
  return (
    vs.find(v => v.lang?.toLowerCase().replace('_', '-') === lower) ??
    vs.find(v => v.lang?.toLowerCase().replace('_', '-').startsWith(base))
  );
}

/**
 * Cihazda verilen dil için ses var mı?
 * Ses listesi henüz yüklenmediyse (boş) true döner — bilinmiyor, engelleme.
 */
export function hasVoiceFor(lang: string): boolean {
  if (allVoices().length === 0) return true;
  return pickVoice(lang) !== undefined;
}

export function initSpeech(): void {
  if (voicesReady) return;
  const load = () => {
    voicesReady = allVoices().length > 0;
  };
  load();
  speechSynthesis.addEventListener('voiceschanged', load);
}

interface Part {
  text: string;
  lang: string;
  rate: number;
}

/** Sıradaki parçaları KUYRUĞA koyar; sıra tarayıcı tarafından yönetilir (zamanlayıcı yok). */
function speakParts(parts: Part[]): void {
  try {
    // Boştayken cancel() çağırmak Chrome'da sonraki speak()'i yutabiliyor
    if (speechSynthesis.speaking || speechSynthesis.pending) speechSynthesis.cancel();
    for (const p of parts) {
      const u = new SpeechSynthesisUtterance(p.text);
      u.lang = p.lang;
      u.rate = p.rate;
      u.pitch = 1;
      const v = pickVoice(p.lang);
      if (v) u.voice = v;
      alive.add(u);
      const done = () => alive.delete(u);
      u.onend = done;
      u.onerror = done;
      speechSynthesis.speak(u);
    }
  } catch {}
}

export function speak(text: string, force = false): void {
  if (!text) return;

  if (!force && text === lastSpokenText) return;

  const now = Date.now();
  if (!force && now - lastSpeak < MIN_GAP_MS) return;

  lastSpeak = now;
  lastFull = text;
  lastSpokenText = text;

  speakParts([{ text, lang: 'tr-TR', rate: 0.9 }]);
  announce(text, 'tr');
}

/** Arapça konuşma — görevliye hitap için. Ekran okuyucuya yazılmaz (üst üste binmesin). */
export function speakArabic(text: string): void {
  if (!text) return;
  speakParts([{ text, lang: 'ar-SA', rate: 0.85 }]);
}

/**
 * Yardım ekranı: önce Arapça (görevliye), SONRA Türkçe (kullanıcıya) — aynı kuyrukta,
 * Arapça bitmeden Türkçe başlamaz ve onu kesmez.
 * @returns Cihazda Arapça ses var mı
 */
export function speakHelp(arabic: string, turkish: string): boolean {
  const parts: Part[] = [];
  if (arabic) parts.push({ text: arabic, lang: 'ar-SA', rate: 0.85 });
  if (turkish) parts.push({ text: turkish, lang: 'tr-TR', rate: 0.9 });
  if (parts.length === 0) return true;
  speakParts(parts);
  return hasVoiceFor('ar-SA');
}

export function repeatLast(): void {
  if (lastFull) speak(lastFull, true);
}
